# 使用三个基因计算 PD 评分（Z-score 标准化后取平均）
core_matrix <- as.matrix(core_expr[, c("IFT20", "DUSP11", "KPNA2")])
core_matrix_scaled <- scale(core_matrix)
PD_score <- rowMeans(core_matrix_scaled, na.rm = TRUE)

core_expr$PD_Score <- PD_score
cat("PD 评分范围:", range(PD_score, na.rm = TRUE), "\n")

plot_data$Group <- factor(plot_data$Group, levels = c("NF", "DCM", "ICM"))

# 1.1 IFT20 三组箱线图
p_ift20 <- ggplot(plot_data, aes(x = Group, y = IFT20, fill = Group)) +
  geom_boxplot(outlier.shape = NA, alpha = 0.7, width = 0.6) +
  geom_jitter(width = 0.15, size = 1.5, alpha = 0.5) +
  stat_compare_means(method = "kruskal.test", 
                     label = "p.format",
                     label.y = max(plot_data$IFT20, na.rm = TRUE) * 1.05,
                     size = 4) +
  stat_compare_means(comparisons = list(c("NF", "DCM"), c("NF", "ICM"), c("DCM", "ICM")),
                     method = "wilcox.test",
                     label = "p.signif",
                     size = 4) +
  labs(title = "IFT20 在 NF/DCM/ICM 中的表达",
       x = "", y = "log2(RPKM+1)") +
  scale_fill_manual(values = c(NF = "#4DAF4A", DCM = "#E41A1C", ICM = "#FF7F00")) +
  theme_minimal(base_size = 14) +
  theme(legend.position = "none",
        plot.title = element_text(hjust = 0.5, face = "bold", size = 16))

# 3. PD 评分与年龄的相关性
# ============================================================

age_data <- data.frame(
  Sample = sample_info_aligned$ID,
  Age = sample_info_aligned$Age,
  PD_Score = PD_score,
  Group = sample_info_aligned$Group,
  stringsAsFactors = FALSE
)

p_age <- ggplot(age_data, aes(x = Age, y = PD_Score, color = Group)) +
  geom_point(size = 2.5, alpha = 0.7) +
  geom_smooth(method = "lm", se = TRUE, color = "black", linetype = "dashed") +

           label.y = max(age_data$PD_Score, na.rm = TRUE) * 0.95,
           size = 4) +
  labs(title = "PD 特征评分与年龄的相关性",
       x = "年龄 (岁)",

# 5. 保存统计结果
# ============================================================

# 三组统计
stats_list <- list()

stats_list$IFT20_3groups <- aggregate(IFT20 ~ Group, data = plot_data, 
                                      FUN = function(x) c(mean = mean(x), sd = sd(x), median = median(x)))

stats_list$DUSP11_3groups <- aggregate(DUSP11 ~ Group, data = plot_data, 
                                       FUN = function(x) c(mean = mean(x), sd = sd(x), median = median(x)))

stats_list$KPNA2_3groups <- aggregate(KPNA2 ~ Group, data = plot_data, 
                                      FUN = function(x) c(mean = mean(x), sd = sd(x), median = median(x)))

stats_list$PD_Score_3groups <- aggregate(PD_Score ~ Group, data = plot_data, 
                                         FUN = function(x) c(mean = mean(x), sd = sd(x), median = median(x)))

# DCM vs IC M Wilcoxon 检验
wilcox_ift20 <- wilcox.test(IFT20 ~ Group, data = dcm_icm_data)
wilcox_dusp11 <- wilcox.test(DUSP11 ~ Group, data = dcm_icm_data)
wilcox_kpna2 <- wilcox.test(KPNA2 ~ Group, data = dcm_icm_data)
wilcox_pd <- wilcox.test(PD_Score ~ Group, data = dcm_icm_data)

stats_list$DCM_vs_ICM_pvalues <- data.frame(
  Gene = c("IFT20", "DUSP11", "KPNA2", "PD_Score"),
  P_Value = c(wilcox_ift20$p.value, wilcox_dusp11$p.value, 
              wilcox_kpna2$p.value, wilcox_pd$p.value)
)

# 年龄相关性
cor_test <- cor.test(age_data$Age, age_data$PD_Score, use = "complete.obs")
stats_list$Age_correlation <- data.frame(
  Method = "Pearson",
  Correlation = cor_test$estimate,
  P_Value = cor_test$p.value
)

write_xlsx(stats_list, "GSE116250_subgroup_statistics.xlsx")
